# PostgreSQL Client for version 17

# Remover o repo errado que foi adicionado (Caso esteja na versão 16)
sudo rm /etc/apt/sources.list.d/pgdg.list

# Adicionar forçando o codinome Ubuntu correto (noble)
sudo sh -c 'echo "deb https://apt.postgresql.org/pub/repos/apt noble-pgdg main" > /etc/apt/sources.list.d/pgdg.list'

# Atualizar e instalar
sudo apt update
sudo apt install -y postgresql-client-17

# Após instalar, verifique:
bash/usr/lib/postgresql/17/bin/pg_dump --version

# E atualize o script com o caminho correto:
bashPG_DUMP="/usr/lib/postgresql/17/bin/pg_dump"
